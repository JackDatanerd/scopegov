Run once from the repo root, in Git Bash; it adds both import lines and both ARTICLES entries, and is safe to run twice:

node - <<'JS'
const fs = require('fs'); const p = 'lib/help/index.ts'; let s = fs.readFileSync(p, 'utf8')
for (const [name, file] of [['guardianSettings','guardian-settings'], ['readYourReports','read-your-reports']]) {
  if (s.includes(name)) { console.log(name + ': already registered'); continue }
  const imports = [...s.matchAll(/^import \{ \w+ \} from '\.\/articles\/[^']+'\n/gm)]; const last = imports[imports.length - 1]
  const at = last.index + last[0].length
  s = s.slice(0, at) + `import { ${name} } from './articles/${file}'\n` + s.slice(at)
  const m = s.match(/export const ARTICLES: HelpArticle\[\] = \[\n([\s\S]*?)\n\]/)
  const end = m.index + m[0].indexOf(m[1]) + m[1].length
  s = s.slice(0, end) + `\n  ${name},` + s.slice(end); console.log(name + ': registered')
}
fs.writeFileSync(p, s)
JS
